// Copy it into place for the one-time install, then delete it again:
//     cp deploy/installers/root-install.php.txt install.php
//     ... finish the wizard at /install.php ...
//     rm install.php
//
// It is a .php.txt file so that a web server can never execute it where it sits,
// and it lives outside the document root so a finished server never carries it.
// The live paths (install.php) are gitignored: a server that has correctly removed the
// installer stays clean, which is what lets CyberPanel's Git Manager pull and push
// at all — it refuses to touch a checkout with uncommitted changes.
// ─────────────────────────────────────────────────────────────────────────────
/**
 * Nowills PMS — Web Installer
 *
 * Access this file in your browser after uploading the app to cPanel:
 *   https://yourdomain.com/install.php
 *
 * It will:
 *  1. Check PHP version and required extensions
 *  2. Let you configure the database
 *  3. Write .env
 *  4. Run migrations
 *  5. Generate APP_KEY
 *  6. Clear cache
 *  7. Delete itself for security
 *
 * After installation completes, delete this file.
 */

// Don't run if already installed (check for a marker file)
if (file_exists(__DIR__.'/storage/framework/installed.php')) {
    http_response_code(404);
    echo '<h1>Installer Already Completed</h1>
    <p>The Nowills PMS has been installed. This installer has been disabled.</p>';
    exit;
}

// ── CSRF + rate-limit ────────────────────────────────────────────────────────
session_start();
if (empty($_SESSION['install_csrf'])) {
    $_SESSION['install_csrf'] = bin2hex(random_bytes(32));
}
$csrfToken = $_SESSION['install_csrf'];
$rateFile = sys_get_temp_dir().'/nowills_root_install_rate_'.md5($_SERVER['REMOTE_ADDR'] ?? 'unknown').'.json';
$rateData = json_decode((string) @file_get_contents($rateFile), true) ?: ['count' => 0, 'reset' => time() + 900];
if (time() > ($rateData['reset'] ?? 0)) {
    $rateData = ['count' => 0, 'reset' => time() + 900];
}
if (($_SERVER['REQUEST_METHOD'] ?? 'GET') === 'POST' && ($rateData['count'] ?? 0) >= 10) {
    http_response_code(429);
    exit(json_encode(['status' => 'error', 'message' => 'Too many attempts. Try again in 15 minutes.']));
}

/** Quote a value for .env — same rules as the public installer. */
function env_quote(string $v): string
{
    return '"'.str_replace(['\\', '"', '$', "\r", "\n"], ['\\\\', '\\"', '\\$', '', ''], $v).'"';
}

// Check if .env exists
$envExists = file_exists(__DIR__.'/.env');

// Handle form submission
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    if (! hash_equals($csrfToken, (string) ($_POST['_csrf'] ?? ''))) {
        http_response_code(419);
        exit(json_encode(['status' => 'error', 'message' => 'Session expired. Reload and try again.']));
    }
    $rateData['count'] = ($rateData['count'] ?? 0) + 1;
    @file_put_contents($rateFile, json_encode($rateData));

    $action = $_POST['action'] ?? '';

    if ($action === 'test_db') {
        $dbHost = $_POST['db_host'] ?? '127.0.0.1';
        $dbPort = $_POST['db_port'] ?? '3306';
        $dbName = $_POST['db_name'] ?? '';
        $dbUser = $_POST['db_user'] ?? '';
        $dbPass = $_POST['db_pass'] ?? '';

        try {
            $pdo = new PDO("mysql:host=$dbHost;port=$dbPort;dbname=$dbName", $dbUser, $dbPass);
            $pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
            echo json_encode(['status' => 'success', 'message' => 'Database connection successful!']);
        } catch (PDOException $e) {
            @error_log('Nowills root installer DB test failed: '.$e->getMessage());
            echo json_encode(['status' => 'error', 'message' => 'Could not connect to MySQL — check host, database, user and password.']);
        }
        exit;
    }

    if ($action === 'install') {
        $dbHost = $_POST['db_host'] ?? '127.0.0.1';
        $dbPort = $_POST['db_port'] ?? '3306';
        $dbName = $_POST['db_name'] ?? '';
        $dbUser = $_POST['db_user'] ?? '';
        $dbPass = $_POST['db_pass'] ?? '';
        $appUrl = $_POST['app_url'] ?? 'http://localhost';
        $appDebug = isset($_POST['app_debug']) ? 'true' : 'false';

        // Reject control characters that would corrupt .env.
        foreach (['database name' => $dbName, 'database user' => $dbUser, 'database password' => $dbPass, 'database host' => $dbHost, 'site URL' => $appUrl] as $label => $v) {
            if (preg_match('/[\x00-\x1F\x7F]/', $v)) {
                exit(json_encode(['status' => 'error', 'message' => "The {$label} contains a line break or control character. Paste it again as a single line."]));
            }
        }

        // Generate APP_KEY
        $key = 'base64:'.base64_encode(random_bytes(32));

        // Read .env.example and write .env — every value goes through env_quote()
        // so a $ or newline in a password can never inject a config line.
        $envContent = file_get_contents(__DIR__.'/.env.example');
        $envContent = str_replace('APP_KEY=', 'APP_KEY='.$key, $envContent);
        $envContent = str_replace('APP_URL=http://localhost:8000', 'APP_URL='.env_quote($appUrl), $envContent);
        $envContent = str_replace('APP_DEBUG=true', 'APP_DEBUG='.$appDebug, $envContent);
        $envContent = str_replace('DB_HOST=127.0.0.1', 'DB_HOST='.env_quote($dbHost), $envContent);
        $envContent = str_replace('DB_PORT=3306', 'DB_PORT='.env_quote($dbPort), $envContent);
        $envContent = str_replace('DB_DATABASE=nowills_pms', 'DB_DATABASE='.env_quote($dbName), $envContent);
        $envContent = str_replace('DB_USERNAME=pms', 'DB_USERNAME='.env_quote($dbUser), $envContent);
        $envContent = str_replace('DB_PASSWORD=change_me', 'DB_PASSWORD='.env_quote($dbPass), $envContent);

        file_put_contents(__DIR__.'/.env', $envContent);
        @chmod(__DIR__.'/.env', 0600);

        // Run migrations
        require __DIR__.'/build/vendor/autoload.php';
        $app = require __DIR__.'/bootstrap/app.php';
        $kernel = $app->make(Illuminate\Contracts\Console\Kernel::class);

        $output = [];
        $exitCode = 0;
        try {
            $exitCode = $kernel->call('migrate');
            $output = $kernel->output();
        } catch (Exception $e) {
            @error_log('Nowills root installer migrate failed: '.$e->getMessage());
            $output = 'Migration failed — check the server error log.';
        }

        // Clear cache
        try {
            $kernel->call('cache:clear');
            $kernel->call('config:clear');
            $kernel->call('view:clear');
            $kernel->call('route:clear');
        } catch (Exception $e) {}

        // Create marker file (only on success)
        if ($exitCode === 0) {
            file_put_contents(__DIR__.'/storage/framework/installed.php', '<?php // Installed: '.date('Y-m-d H:i:s'));
            // Self-delete so the installer can never be replayed.
            @unlink(__FILE__);
        }

        // Output result — never echo raw migration output / exception text.
        if ($exitCode === 0) {
            echo json_encode(['status' => 'success', 'message' => 'Installation complete! Delete this file now.']);
        } else {
            echo json_encode(['status' => 'error', 'message' => 'Migration failed — check the server error log.']);
        }
        exit;
    }
}

// Check PHP version
if (version_compare(PHP_VERSION, '8.3.0', '<')) {
    echo '<div style="background:#fee;padding:20px;border:1px solid #c33;border-radius:8px;margin:20px;">
        <h2>PHP Version Error</h2>
        <p>This application requires PHP 8.3 or higher. You are running PHP '.PHP_VERSION.'.</p>
        <p>In cPanel, go to <strong>MultiPHP Manager</strong> and select PHP 8.3 or 8.4 for your domain.</p>
    </div>';
    exit;
}

// Check required extensions
$required = ['pdo_mysql', 'mbstring', 'openssl', 'tokenizer', 'ctype', 'json', 'dom', 'xml', 'fileinfo'];
$missing = [];
foreach ($required as $ext) {
    if (!extension_loaded($ext)) {
        $missing[] = $ext;
    }
}
if (!empty($missing)) {
    echo '<div style="background:#fee;padding:20px;border:1px solid #c33;border-radius:8px;margin:20px;">
        <h2>Missing Extensions</h2>
        <p>The following PHP extensions are required but not installed:</p>
        <ul><li>'.implode('</li><li>', $missing).'</li></ul>
        <p>In cPanel, go to <strong>MultiPHP INI Editor</strong> or contact your host to enable them.</p>
    </div>';
    exit;
}

<script>
const testBtn = document.getElementById('test-btn');
const installBtn = document.getElementById('install-btn');
const loading = document.getElementById('loading');
const loadingText = document.getElementById('loading-text');
const result = document.getElementById('result');

function showLoading(text) {
    loadingText.textContent = text;
    loading.classList.add('show');
}

function hideLoading() {
    loading.classList.remove('show');
}

function showResult(status, message) {
    result.innerHTML = '<div class="result ' + status + '">' + message + '</div>';
}

testBtn.addEventListener('click', async () => {
    const formData = new FormData(document.getElementById('install-form'));
    formData.append('action', 'test_db');

    showLoading('Testing database connection...');
    testBtn.disabled = true;

    try {
        const response = await fetch('', { method: 'POST', body: formData });
        const data = await response.json();
        if (data.status === 'success') {
            showResult('success', '✓ ' + data.message);
        } else {
            showResult('error', '✗ ' + data.message);
        }
    } catch (e) {
        showResult('error', '✗ Request failed: ' + e.message);
    }

    hideLoading();
    testBtn.disabled = false;
});

installBtn.addEventListener('click', async () => {
    if (!confirm('This will write .env, run migrations, and clear cache. Continue?')) return;

    const formData = new FormData(document.getElementById('install-form'));
    formData.append('action', 'install');

    showLoading('Writing .env...');
    installBtn.disabled = true;
    testBtn.disabled = true;

    try {
        const response = await fetch('', { method: 'POST', body: formData });
        const data = await response.json();
        if (data.status === 'success') {
            showResult('success', '✓ ' + data.message + ' — You can now visit your site!');
            document.getElementById('step-db').classList.add('done');
            document.getElementById('step-db').classList.remove('active');
            document.getElementById('step-install').classList.add('done');
            document.getElementById('step-install').classList.remove('active');
            document.getElementById('step-done').classList.add('active');
        } else {
            showResult('error', '✗ ' + data.message + (data.output ? '<br><pre>' + data.output + '</pre>' : ''));
        }
    } catch (e) {
        showResult('error', '✗ Request failed: ' + e.message);
    }

    hideLoading();
    installBtn.disabled = false;
    testBtn.disabled = false;
});
</script>
